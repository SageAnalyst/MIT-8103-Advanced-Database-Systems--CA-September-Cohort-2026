2. MYSQL ENGINE PARAMETER GROUP TUNING (my.cnf / RDS Parameter Group)




# Binary logging configuration for cross-replica replication
binlog_format                   = ROW
binlog_row_image                = FULL
expire_logs_days                = 7
sync_binlog                     = 1        # Strict durability: flush to disk on every commit


# InnoDB memory pool allocation and buffer sizing
innodb_buffer_pool_size         = 8G       # Allocated ~70-80% of dedicated RAM on db.m6g.xlarge
innodb_buffer_pool_instances   = 8
innodb_log_file_size            = 1G       # Write-ahead redo log sizing
innodb_flush_log_at_trx_commit  = 1        # Enforces ACID durability compliance on Master
innodb_flush_method             = O_DIRECT


# Concurrency and thread handling
max_connections                 = 500
wait_timeout                    = 28800
interactive_timeout             = 28800
lock_wait_timeout               = 50       # Fails deadlocked queries after 50 seconds
innodb_lock_wait_timeout        = 10       # Drops blocked locks in pessimistic test scenarios


# Read-Replica safety constraints
read_only                       = 1        # Enabled exclusively on read replica nodes
super_read_only                 = 1        # Enforces read-only behavior for administrative roles
